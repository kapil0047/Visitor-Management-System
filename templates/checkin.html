from flask import Flask, render_template, request, redirect import psycopg2 import os from datetime import datetime app = Flask(__name__) app.config['UPLOAD_FOLDER'] = 'static/uploads' # Connect to PostgreSQL def get_db_connection(): return psycopg2.connect( dbname='visitor_db', user='postgres', password='your_password', # replace with your password host='localhost', port='5432' ) @app.route('/') def welcome(): return render_template('welcome.html') @app.route('/checkin', methods=['GET', 'POST']) def checkin(): conn = get_db_connection() cur = conn.cursor() if request.method == 'POST': name = request.form['name'] email = request.form['email'] phone = request.form['phone'] purpose = request.form['purpose'] employee_id = request.form['employee_id'] photo = request.files['photo'] if photo: photo_path = os.path.join(app.config['UPLOAD_FOLDER'], photo.filename) photo.save(photo_path) else: photo_path = None cur.execute(""" INSERT INTO visitor (name, email, phone, purpose, photo, employee_id, checkin) VALUES (%s, %s, %s, %s, %s, %s, %s) """, (name, email, phone, purpose, photo_path, employee_id, datetime.now())) conn.commit() cur.close() conn.close() return render_template('success.html', message="Check-in successful!") # Fetch employees for dropdown cur.execute("SELECT id, name FROM employee") employees = cur.fetchall() cur.close() conn.close() return render_template('checkin.html', employees=employees) if __name__ == '__main__': app.run(debug=True)
